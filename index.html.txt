<script>
const $ = id => document.getElementById(id);

const products = [
{name:"뮤이 팔찌",img:"images/01.png"},
{name:"팔찌 2",img:"images/02.png"},
{name:"오니크 팔찌",img:"images/03.png"},
{name:"큐티 아모르 팔찌",img:"images/04.png"},
{name:"블링 팔찌",img:"images/05.png"},
{name:"아모르 팔찌",img:"images/06.png"},
{name:"러블리아 팔찌",img:"images/07.png"},
{name:"아몬드 팔찌",img:"images/08.png"},
{name:"러브레터 팔찌",img:"images/09.png"},
{name:"플로리안 팔찌",img:"images/10.png"},
{name:"로맨틱 포인트 팔찌",img:"images/11.png"},
{name:"사랑해 팔찌",img:"images/12.png"},
{name:"클리프투게더 클립체인 팔찌",img:"images/13.png"},
{name:"팔찌 14",img:"images/14.png"},
{name:"팔찌 15",img:"images/15.png"},
{name:"팔찌 16",img:"images/16.png"}
];

let current = [];
let next = [];
let matchIndex = 0;

function shuffle(arr){
    for(let i=arr.length-1;i>0;i--){
        const j=Math.floor(Math.random()*(i+1));
        [arr[i],arr[j]]=[arr[j],arr[i]];
    }
}
function startGame(){

    current=[...products];
    shuffle(current);

    next=[];
    matchIndex=0;

    $("start").style.display="none";
    $("game").style.display="block";

    showMatch();
}

function showMatch(){

    // 우승
    if(current.length===1){

        $("game").style.display="none";
        $("winner").style.display="block";

        $("winnerImg").src=current[0].img;
        $("winnerName").innerText=current[0].name;

        return;
    }

    // 현재 라운드 종료
    if(matchIndex>=current.length){

        current=[...next];
        next=[];
        matchIndex=0;

        shuffle(current);
    }

    // 홀수 자동 진출
    if(matchIndex===current.length-1){

        next.push(current[matchIndex]);
        matchIndex++;

        showMatch();
        return;
    }

    const left=current[matchIndex];
    const right=current[matchIndex+1];

    $("leftImg").src=left.img;
    $("leftName").innerText=left.name;

    $("rightImg").src=right.img;
    $("rightName").innerText=right.name;

    $("round").innerText=current.length+"강";
    $("progress").innerText=(matchIndex/2+1)+" / "+Math.ceil(current.length/2);
}
function pick(side){

    if(side===0){
        next.push(current[matchIndex]);
    }else{
        next.push(current[matchIndex+1]);
    }

    matchIndex+=2;

    showMatch();
}

// Enter 키로 시작
document.addEventListener("keydown",e=>{
    if(e.key==="Enter" && $("start").style.display!=="none"){
        startGame();
    }
});

// 이미지가 없으면 표시
document.querySelectorAll("img").forEach(img=>{
    img.onerror=()=>{
        img.src="https://placehold.co/600x600?text=No+Image";
    };
});
